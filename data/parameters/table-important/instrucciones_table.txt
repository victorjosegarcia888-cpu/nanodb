La forma más simple de motor líquido es el sistema presurizado (pressure‑fed).
La forma más eficiente es el ciclo de combustión escalonada, especialmente el full‑flow staged combustion (como el Raptor).

Esto coincide con la clasificación oficial de NASA sobre motores líquidos:

Simple → Pressure‑fed

Eficiente → Pump‑fed + Staged combustion

🧪 Motores líquidos: simple vs eficiente
🟢 1. El más simple: Pressure‑Fed (alimentación por presión)
Cómo funciona:  
Los tanques se presurizan con helio o nitrógeno y empujan el propelente hacia la cámara.

Ventajas:

Sin turbobombas → ultra simple.

Fiable para maniobra orbital.

Barato y fácil de fabricar.

Desventajas:

Baja presión de cámara → bajo rendimiento.

Tanques pesados → no sirve para lanzadores.

Aplicaciones típicas:

Motores de satélites.

Maniobra orbital.

Upper stages pequeños.

Fuente NASA: Clasificación oficial de sistemas de alimentación líquida.

🔵 2. El más eficiente: Pump‑Fed + Staged Combustion
presión del propelente y un preburner para quemar parte del combustible y accionar la turbobomba.

Ventajas:

Máxima presión de cámara.

Máxima eficiencia (Isp).

Ideal para motores de primera etapa y reutilizables.

Desventajas:

Complejidad extrema.

Requiere materiales avanzados (UHTC, superaleaciones, cuasicristales).

Ejemplos:

RS‑25 (NASA)

Raptor (SpaceX)

RD‑180 / RD‑191 (Rusia)

Fuente NASA: Diseño de motores líquidos y cámaras de combustión.

public static class TaskMotorCompleto

    {

        var tobera = TaskTobera.CrearTobera(ctx);
        var inyector = TaskInyector.CrearInyector(ctx);
        var turbobombaLOX = TaskTurbobombaLOX.CrearTurbobomba(ctx);

        var preburnerOX = TaskPreburnerOX.CrearPreburner(ctx);
        var preburnerFuel = TaskPreburnerFuel.CrearPreburner(ctx);

        var motor = ctx.BooleanUnion(camara, tobera);
        motor = ctx.BooleanUnion(motor, inyector);
        motor = ctx.BooleanUnion(motor, turbobombaLOX);
        motor = ctx.BooleanUnion(motor, turbobombaCH4);
        motor = ctx.BooleanUnion(motor, preburnerOX);
        motor = ctx.BooleanUnion(motor, preburnerFuel);

        ctx.Show(motor);
    }
}

using Leap71.ShapeKernel;

public static class TaskInyector
{
    public static Geometry CrearInyector(PicoGKContext ctx)
    {
        float radioPlaca = 0.5f;
        var placa = ctx.SolidCylinder(radioPlaca, 0.02f);

        // Orificios simplificados: patrón circular
        for (int i = 0; i < DatosMotor.NumeroOrificios; i += 100)
        {
            float angle = (float)(2 * Math.PI * i / DatosMotor.NumeroOrificios);
            float r = radioPlaca * 0.8f;
            float x = r * MathF.Cos(angle);
            float y = r * MathF.Sin(angle);

            var orificio = ctx.SolidCylinder(DatosMotor.DiametroOrificio * 0.001f, 0.03f);
            orificio = ctx.Translate(orificio, x, y, 0);

            placa = ctx.BooleanDifference(placa, orificio);
        }

        return placa;
    }
}

using Leap71.ShapeKernel;

public static class TaskTobera
{
    public static Geometry CrearTobera(PicoGKContext ctx)
    {
        // Cuello según NASA: Dt ≈ sqrt(c* / Pc)
        float diametroCuello = MathF.Sqrt(DatosMotor.cStar / DatosMotor.Pc) * 0.6f;

        // Área de salida Ae = At * Ae/At
        float diametroSalida = diametroCuello * MathF.Sqrt(DatosMotor.Ae_At);

        // Longitud típica NASA: 0.8–1.2 × Dt
        float longitud = diametroCuello * 1.2f;

        // Convergente + divergente (contorno Rao simplificado)
        var cuello = ctx.SolidCylinder(diametroCuello, longitud * 0.2f);
        var divergente = ctx.SolidCone(diametroCuello, diametroSalida, longitud * 0.8f);

        var tobera = ctx.BooleanUnion(cuello, divergente);

        // Liner UHTC + cuasicristal
        var linerUHTC = ctx.Shell(tobera, 0.004f);
        var linerQC = ctx.Shell(linerUHTC, 0.002f);

        return linerQC;
    }
}

csharp
using Leap71.ShapeKernel;

public static class TaskCanalesRegenerativos
{
    public static Geometry CrearCanales(PicoGKContext ctx, Geometry toberaBase)
    {
        // Parámetros NASA típicos
        float ancho = 0.002f;     // 2 mm
        float altura = 0.003f;    // 3 mm
        float pitch = 0.004f;     // 4 mm
        float espesorPared = 0.001f; // 1 mm

        // Extraer bounding box de la tobera
        var bb = ctx.GetBoundingBox(toberaBase);
        float largo = bb.LengthZ;

        Geometry canales = null;

        // Crear canales alrededor de la tobera
        for (float z = 0; z < largo; z += pitch)
        {
            var canal = ctx.SolidBox(bb.LengthX, ancho, altura);
            canal = ctx.Translate(canal, 0, 0, z);

            canales = canales == null ? canal : ctx.BooleanUnion(canales, canal);
        }

        // Restar canales del cuerpo de la tobera
        var toberaConCanales = ctx.BooleanDifference(toberaBase, canales);

        // Añadir pared interna UHTC/cuasicristal
        var paredInterna = ctx.Shell(toberaConCanales, espesorPared);

        return paredInterna;
    }
}

using Leap71.ShapeKernel;

public static class TaskContornoRao
{
    public static Geometry CrearContornoRao(PicoGKContext ctx)
    {
        float Dt = MathF.Sqrt(DatosMotor.cStar / DatosMotor.Pc) * 0.6f;   // diámetro cuello
        float De = Dt * MathF.Sqrt(DatosMotor.Ae_At);                      // diámetro salida
        float L = Dt * 1.2f;                                               // longitud típica NASA

        // Perfil Rao simplificado: curva suave entre Dt y De
        var puntos = new List<Vec3>();

        for (float z = 0; z <= L; z += L / 50f)
        {
            float t = z / L;
            float diam = Dt + (De - Dt) * (3 * t * t - 2 * t * t * t);     // interpolación cúbica Rao
            puntos.Add(new Vec3(diam, 0, z));
        }

        var contorno = ctx.SolidRevolve(puntos);

        // Liner UHTC + cuasicristal
        var linerUHTC = ctx.Shell(contorno, 0.004f);
        var linerQC = ctx.Shell(linerUHTC, 0.002f);

        return linerQC;
    }
}

using Leap71.ShapeKernel;

public static class TaskConductosGasFFSC
{
    public static Geometry CrearConductos(PicoGKContext ctx)
    {
        float radioConducto = 0.05f;     // 5 cm típico NASA
        float longitud = 0.6f;           // longitud típica entre preburner y turbina

        // Conducto OX-rich
        var ox = ctx.SolidCylinder(radioConducto, longitud);
        ox = ctx.Translate(ox, 0.2f, 0, 0);

        // Conducto Fuel-rich
        var fuel = ctx.SolidCylinder(radioConducto * 0.9f, longitud);
        fuel = ctx.Translate(fuel, -0.2f, 0, 0);

        // Unión hacia cámara
        var union = ctx.SolidCone(radioConducto, radioConducto * 1.5f, 0.3f);
        union = ctx.Translate(union, 0, 0, longitud);

        var conductos = ctx.BooleanUnion(ox, fuel);
        conductos = ctx.BooleanUnion(conductos, union);

        // Coating cuasicristal para reducir conductividad térmica
        var linerQC = ctx.Shell(conductos, 0.003f);

        return linerQC;
    }
}
🟦 TaskLatticeInterno.cs
Estructura interna tipo lattice para:

csharp
using Leap71.ShapeKernel;

public static class TaskLatticeInterno
{
    public static Geometry CrearLattice(PicoGKContext ctx, Geometry baseGeo)
    {
        var bb = ctx.GetBoundingBox(baseGeo);

        float cell = 0.05f;     // tamaño de celda
        float radioBarra = 0.005f;

        Geometry lattice = null;

        for (float x = bb.MinX; x < bb.MaxX; x += cell)
        {
            for (float y = bb.MinY; y < bb.MaxY; y += cell)
            {
                for (float z = bb.MinZ; z < bb.MaxZ; z += cell)
                {
                    var barra = ctx.SolidCylinder(radioBarra, cell);
                    barra = ctx.Rotate(barra, 45, 45, 0);
                    barra = ctx.Translate(barra, x, y, z);

                    lattice = lattice == null ? barra : ctx.BooleanUnion(lattice, barra);
                }
            }
        }

        // Intersección con la geometría base
        var latticeFinal = ctx.BooleanIntersection(baseGeo, lattice);

        return latticeFinal;
    }
}

public static class TaskTurbinaFFSC
{
    public static Geometry CrearTurbina(PicoGKContext ctx)
    {
        float radioRotor = 0.12f;     // típico NASA
        float anchoRotor = 0.05f;
        float numAlabes = 20;

        // Rotor
        var rotor = ctx.SolidCylinder(radioRotor, anchoRotor);

        // Álabes FFSC
        Geometry alabes = null;

        for (int i = 0; i < numAlabes; i++)
        {
            float angle = (float)(i * (360.0 / numAlabes));

        }

        var turbina = ctx.BooleanUnion(rotor, alabes);

        // Coating cuasicristal para alta temperatura
        var linerQC = ctx.Shell(turbina, 0.002f);

public static class TaskInyectorCoaxial
{
    public static Geometry CrearInyector(PicoGKContext ctx)
    {
        float radioPlaca = 0.4f;
        var placa = ctx.SolidCylinder(radioPlaca, 0.02f);

        int numCoaxiales = DatosMotor.NumeroOrificios / 20;

        for (int i = 0; i < numCoaxiales; i++)
        {
            float angle = (float)(i * (360.0 / numCoaxiales));
            float r = radioPlaca * 0.7f;

            float x = r * MathF.Cos(angle);
            float y = r * MathF.Sin(angle);

            // Orificio central (combustible)
            var fuel = ctx.SolidCylinder(DatosMotor.DiametroOrificio * 0.001f, 0.03f);
            fuel = ctx.Translate(fuel, x, y, 0);

            // Orificio anular (oxidante)
            var ox = ctx.SolidCylinder(DatosMotor.DiametroOrificio * 0.002f, 0.03f);
            ox = ctx.Translate(ox, x, y, 0);

            placa = ctx.BooleanDifference(placa, fuel);
            placa = ctx.BooleanDifference(placa, ox);
        }

        return placa;
    }
}

public static class TaskPreburnerCompleto
{
    public static Geometry CrearPreburner(PicoGKContext ctx, bool oxRich)
    {
        float radio = oxRich ? 0.22f : 0.18f;
        float longitud = oxRich ? 0.45f : 0.35f;

        var cuerpo = ctx.SolidCylinder(radio, longitud);

        // Inyector interno del preburner
        var inyector = ctx.SolidCylinder(radio * 0.4f, longitud * 0.2f);
        inyector = ctx.Translate(inyector, 0, 0, longitud * 0.1f);

        // Cámara de mezcla
        var mezcla = ctx.SolidCylinder(radio * 0.6f, longitud * 0.3f);
        mezcla = ctx.Translate(mezcla, 0, 0, longitud * 0.3f);

        // Salida hacia turbina
        var salida = ctx.SolidCone(radio * 0.6f, radio * 0.3f, longitud * 0.3f);
        salida = ctx.Translate(salida, 0, 0, longitud * 0.6f);

        var preburner = ctx.BooleanUnion(cuerpo, inyector);
        preburner = ctx.BooleanUnion(preburner, mezcla);
        preburner = ctx.BooleanUnion(preburner, salida);

        // Liner cuasicristal para alta temperatura
        var linerQC = ctx.Shell(preburner, 0.003f);

        return linerQC;
    }
}

public static class TaskContornoRaoAvanzado
{
    public static Geometry CrearToberaRao(object ctx)
    {
        float Dt = MathF.Sqrt(DatosMotor.cStar / DatosMotor.Pc) * 0.6f;   // cuello
        float De = Dt * MathF.Sqrt(DatosMotor.Ae_At);                      // salida
        float L = Dt * 1.2f;                                               // longitud

        var puntos = new List<Vec3>();

        // Perfil Rao avanzado: suavizado cúbico
        for (float z = 0; z <= L; z += L / 100f)
        {
            float t = z / L;
            // ecuación tipo Rao: transición suave
            float diam = Dt + (De - Dt) * (10 * t * t * t - 15 * t * t * t * t + 6 * t * t * t * t * t);
            puntos.Add(new Vec3(diam, 0, z));
        }

        var tobera = ctx.SolidRevolve(puntos);

        // Liner UHTC + cuasicristal
        var linerUHTC = ctx.Shell(tobera, 0.004f);
        var linerQC = ctx.Shell(linerUHTC, 0.002f);

        return linerQC;
    }
}

public static class TaskCanalesRegenerativosAvanzados
{
    public static Geometry CrearCanales(object ctx, Geometry toberaBase)
    {
        float ancho = 0.002f;        // 2 mm
        float altura = 0.003f;       // 3 mm
        float pitch = 0.004f;        // 4 mm

        var bb = ctx.GetBoundingBox(toberaBase);
        float largo = bb.LengthZ;

        Geometry canales = null;

        // Patrón tipo “lattice” de canales
        for (float z = 0; z < largo; z += pitch)
        {
            for (float angle = 0; angle < 360; angle += 60)
            {
                float r = bb.LengthX * 0.4f;
                float x = r * MathF.Cos(angle * MathF.PI / 180f);
                float y = r * MathF.Sin(angle * MathF.PI / 180f);

                var canal = ctx.SolidBox(ancho, altura, pitch);
                canal = ctx.Translate(canal, x, y, z);

                canales = canales == null ? canal : ctx.BooleanUnion(canales, canal);
            }
        }

        // Restar canales de la tobera
        var toberaConCanales = ctx.BooleanDifference(toberaBase, canales);

        // Pared interna UHTC + cuasicristal
        var paredInterna = ctx.Shell(toberaConCanales, 0.001f);

        return paredInterna;
    }
}
