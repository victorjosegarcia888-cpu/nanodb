🧠 11. Extensión para TasksRaptor.cs (Disociación + OTDF + Coking)
csharp
public class TasksRaptor {
    public double CalcularDisociacion(double T, double P) {
        // Basado en UC3M Fig. 2.5
        return 0.15 + 0.0001 * (T - 1500);
    }

    public double CalcularOTDF(double Tmax, double Tavg, double Tinlet) {
        return (Tmax - Tavg) / (Tavg - Tinlet);
    }

    public double CalcularRiesgoCoking(double T) {
        if (T < 850) return 0.1;
        if (T <= 1200) return 0.85;
        return 0.3;
    }
}
